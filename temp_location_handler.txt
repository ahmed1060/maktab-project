    const handleCaptureLocation = async () => {
        if (!navigator.geolocation) {
            setLocationStatus('Geolocation not supported')
            return
        }

        setCapturingLocation(true)
        setLocationStatus('Capturing location...')

        navigator.geolocation.getCurrentPosition(
            async (position) => {
                const { latitude, longitude } = position.coords
                const coordString = `${latitude.toFixed(6)}, ${longitude.toFixed(6)}`
                
                let capturedLocation = coordString
                
                // Try reverse geocoding
                try {
                    const response = await fetch(
                        `https://nominatim.openstreetmap.org/reverse?format=json&lat=${latitude}&lon=${longitude}`
                    )
                    const data = await response.json()
                    capturedLocation = data.display_name || coordString
                } catch (error) {
                    // Fallback to coordinates
                    capturedLocation = coordString
                }
                
                setLocation(capturedLocation)
                setLocationStatus('Location captured ✓ Saving...')
                
                // Auto-save draft with location
                try {
                    await fetch(`/api/tasks/${task.id}/submit`, {
                        method: 'PUT',
                        body: JSON.stringify({
                            newAttachments: [],
                            note: '',
                            cost: 0,
                            costDescription: '',
                            location: capturedLocation,
                            isDraft: true
                        }),
                        headers: { 'Content-Type': 'application/json' }
                    })
                    
                    setLocationStatus('Location saved ✓')
                    setLocation('') // Clear for next capture
                    router.refresh()
                    
                    setTimeout(() => {
                        setLocationStatus('')
                    }, 3000)
                } catch (error) {
                    console.error('Auto-save failed:', error)
                    setLocationStatus('Location captured but save failed')
                }
                
                setCapturingLocation(false)
            },
            (error) => {
                console.error('Geolocation error:', error)
                setLocationStatus('Failed to get location. Please enable location access.')
                setCapturingLocation(false)
            },
            { enableHighAccuracy: true, timeout: 10000, maximumAge: 0 }
        )
    }
